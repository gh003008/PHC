```python
"""Compute natural forward velocity (v_x) for every clip in a motion pkl.
Outputs: stdout table + JSON file at sample_data/clip_speeds_<source>.json.

Usage: python scripts/data/compute_clip_speeds.py sample_data/amass_isaac_walking_primitive.pkl
"""
import argparse
import json
import os
import sys

import joblib
import numpy as np


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("pkl_path")
    ap.add_argument("--fps", type=int, default=30)
    ap.add_argument("--min_v", type=float, default=0.30,
                    help="filter forward-walking only")
    ap.add_argument("--min_frames", type=int, default=60)
    ap.add_argument("--max_frames", type=int, default=250)
    args = ap.parse_args()

    data = joblib.load(args.pkl_path)
    print(f"Loaded {len(data)} clips from {args.pkl_path}")
    print(f"{'clip_name':<60s} {'frames':>6s} {'v_x_nat (m/s)':>15s}")
    print("-" * 85)

    rows = []
    for name, clip in data.items():
        if "trans_orig" in clip:
            trans = clip["trans_orig"]
        elif "root_trans_offset" in clip:
            trans = clip["root_trans_offset"]
            if hasattr(trans, "cpu"):
                trans = trans.cpu().numpy()
        else:
            continue

        T = trans.shape[0]
        if T < args.min_frames or T > args.max_frames:
            continue

        v_x = float((trans[T - 1, 0] - trans[0, 0]) / (T / args.fps))
        if v_x < args.min_v:
            continue

        rows.append({"name": name, "frames": int(T), "v_x_nat": v_x})
        print(f"{name:<60s} {T:>6d} {v_x:>15.3f}")

    rows.sort(key=lambda r: r["v_x_nat"])
    out_path = os.path.splitext(args.pkl_path)[0] + "_speeds.json"
    with open(out_path, "w") as f:
        json.dump(rows, f, indent=2)
    print(f"\nWrote {out_path} ({len(rows)} forward-walking clips)")


if __name__ == "__main__":
    main()
```

### Task 2: Pick 3 speed-spaced clips and save new pkl

**Files:**

- [ ] **Step 1: Write the picker script**

```python
"""Pick 3 forward-walking clips spaced at ~0.45 / ~0.58 / ~0.72 m/s
and save as a new motion pkl.

Usage:
  python scripts/data/pick_3clips_speedspaced.py
"""
import json
import os

import joblib

SOURCE_PKL = "sample_data/amass_isaac_walking_primitive.pkl"
SPEEDS_JSON = "sample_data/amass_isaac_walking_primitive_speeds.json"
TARGET_SPEEDS = [0.45, 0.58, 0.72]
OUT_PKL = "sample_data/amass_walking_3clips_speedspaced.pkl"


def closest_clip(rows, target_v):
    best = None
    best_diff = float("inf")
    for r in rows:
        d = abs(r["v_x_nat"] - target_v)
        if d < best_diff:
            best_diff = d
            best = r
    return best


def main():
    with open(SPEEDS_JSON) as f:
        rows = json.load(f)
    print(f"Loaded {len(rows)} candidate clips from {SPEEDS_JSON}")

    picked_names = []
    for tv in TARGET_SPEEDS:
        c = closest_clip(rows, tv)
        if c is None:
            raise RuntimeError(f"No clip near {tv} m/s")
        if c["name"] in picked_names:
            # avoid duplicate; pick second-closest
            rows_remaining = [r for r in rows if r["name"] not in picked_names]
            c = closest_clip(rows_remaining, tv)
        picked_names.append(c["name"])
        print(f"  target {tv:.2f} -> {c['name']} (frames={c['frames']}, v_x={c['v_x_nat']:.3f})")

    source = joblib.load(SOURCE_PKL)
    subset = {name: source[name] for name in picked_names}

    joblib.dump(subset, OUT_PKL)
    size_mb = os.path.getsize(OUT_PKL) / 1024 / 1024
    print(f"\nWrote {OUT_PKL} ({size_mb:.1f} MB, {len(subset)} clips)")


if __name__ == "__main__":
    main()
```

```python
"""Make a 60s seamless walking clip by extracting one gait cycle from KIT_11
and looping it phase-aligned.

Detection: heel-strikes via root z-velocity zero-crossings (going positive).
Loop construction: take cycle from HS_i to HS_j, repeat until target duration.
Pose continuity: smooth pose interpolation across loop boundaries with
short blend window (5 frames).

Usage: python scripts/data/make_seamless_long_clip.py
"""
import numpy as np
import joblib

SOURCE_PKL = "sample_data/amass_isaac_walking_forward_single.pkl"
OUT_PKL = "sample_data/amass_walking_kit11_seamless_60s.pkl"
TARGET_DURATION_S = 60.0
FPS = 30
BLEND_FRAMES = 5  # smoothing window across loop joins


def detect_heel_strikes(root_trans, fps=30):
    """Return frame indices of heel-strike events via root vertical
    velocity sign-change (going up after being down, i.e. low point)."""
    z = root_trans[:, 2]
    vz = np.diff(z, prepend=z[0])
    # sign-change from negative to positive = local minimum of z = heel strike
    hs = np.where((vz[:-1] < 0) & (vz[1:] >= 0))[0] + 1
    return hs


def build_seamless_loop(clip, n_frames_target):
    pose_quat = np.asarray(clip["pose_quat_global"])     # (T, 24, 4)
    pose_quat_local = np.asarray(clip["pose_quat"])       # (T, 24, 4)
    pose_aa = np.asarray(clip["pose_aa"])                 # (T, 72)
    trans = np.asarray(clip["trans_orig"])                # (T, 3)

    T = trans.shape[0]
    hs = detect_heel_strikes(trans, FPS)
    print(f"  detected {len(hs)} heel strikes at frames {hs.tolist()}")

    if len(hs) < 2:
        raise RuntimeError("not enough heel strikes for cycle extraction")

    # use consecutive HS pair from middle for cleanest cycle (~1 stride)
    mid_idx = len(hs) // 2
    hs_a, hs_b = hs[max(0, mid_idx - 1)], hs[mid_idx]
    cycle_len = hs_b - hs_a
    print(f"  cycle: frames [{hs_a}, {hs_b}) len={cycle_len}")

    cycle_pq = pose_quat[hs_a:hs_b].copy()
    cycle_pql = pose_quat_local[hs_a:hs_b].copy()
    cycle_aa = pose_aa[hs_a:hs_b].copy()
    cycle_tr = trans[hs_a:hs_b].copy()

    # offset trans so cycle starts at origin
    cycle_tr -= cycle_tr[0]
    cycle_dx = cycle_tr[-1] - cycle_tr[0]  # how much root advances per cycle

    n_loops = int(np.ceil(n_frames_target / cycle_len))
    out_pq = np.zeros((cycle_len * n_loops, 24, 4), dtype=cycle_pq.dtype)
    out_pql = np.zeros((cycle_len * n_loops, 24, 4), dtype=cycle_pql.dtype)
    out_aa = np.zeros((cycle_len * n_loops, 72), dtype=cycle_aa.dtype)
    out_tr = np.zeros((cycle_len * n_loops, 3), dtype=cycle_tr.dtype)

    for i in range(n_loops):
        s = i * cycle_len
        e = s + cycle_len
        out_pq[s:e] = cycle_pq
        out_pql[s:e] = cycle_pql
        out_aa[s:e] = cycle_aa
        out_tr[s:e] = cycle_tr + i * cycle_dx

    out_pq = out_pq[:n_frames_target]
    out_pql = out_pql[:n_frames_target]
    out_aa = out_aa[:n_frames_target]
    out_tr = out_tr[:n_frames_target]

    return {
        "pose_quat_global": out_pq,
        "pose_quat": out_pql,
        "pose_aa": out_aa,
        "trans_orig": out_tr,
        "root_trans_offset": out_tr.copy(),
        "beta": clip["beta"],
        "gender": clip["gender"],
        "fps": FPS,
    }


def main():
    src = joblib.load(SOURCE_PKL)
    name, clip = next(iter(src.items()))
    print(f"Source: {name} ({clip['pose_quat_global'].shape[0]} frames)")

    n_frames = int(TARGET_DURATION_S * FPS)
    out_clip = build_seamless_loop(clip, n_frames)
    out = {f"{name}_seamless_60s": out_clip}
    joblib.dump(out, OUT_PKL)
    print(f"\nWrote {OUT_PKL} (1 clip, {n_frames} frames = {n_frames/FPS:.1f}s)")


if __name__ == "__main__":
    main()
```
